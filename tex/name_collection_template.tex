%   %%WORKSHEET_ID%%   -> \WorksheetSetup{worksheet id=<id>} (or empty)
%   %%STUDENT_NAME%%   -> the target name, e.g. Jane Doe
%
% Compile in cv mode (red, extractable grid boxes) or blank mode (plain boxes
% for the student), the same as the other worksheets:
%   latexmk -pdf -usepretex='\def\WSCVMode{1}' name_collection_template.tex
